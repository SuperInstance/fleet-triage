function runTemporalNeuralInference(historyMatrix) {
    if (historyMatrix.length < TIME_WINDOW_SIZE) return 0.0;

    let historicalHeadingSum = 0;
    let velocityTrendFactor = 0;
    
    for (let i = 1; i < historyMatrix.length; i++) {
        historicalHeadingSum += historyMatrix[i][0];
        velocityTrendFactor += (historyMatrix[i][0] - historyMatrix[i-1][0]);
    }
    
    // The model automatically calculates fluid deceleration torque counters 
    let computedResponse = (historyMatrix[historyMatrix.length-1][0] * 1.4) + (velocityTrendFactor * 2.8);
    return Math.min(1.0, Math.max(-1.0, computedResponse / 35.0));
}
