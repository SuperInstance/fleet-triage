# prepare.pyimport torchimport numpy as npimport timeimport threadingfrom dataclasses import dataclassfrom collections import deque

@dataclassclass SensorMapping:
    name: str
    index: int
    min_physical: float
    max_physical: float
    unit: str
QUILT_SCHEMA = {
    "heading_error": SensorMapping("Heading Error", 0, -180.0, 180.0, "degrees"),
    "xte":           SensorMapping("Cross-Track Error", 1, -50.0, 50.0, "meters"),
    "rudder_angle":  SensorMapping("Current Rudder Position", 2, -35.0, 35.0, "degrees"),
    "roll":          SensorMapping("Vessel Roll", 3, -45.0, 45.0, "degrees"),
    "speed":         SensorMapping("Vessel Speed", 4, 0.0, 30.0, "knots"),
    "target_pump":   SensorMapping("Hydraulic Pump Command", 5, -1.0, 1.0, "voltage_scalar")
}
class QuiltVesselStateMatrix:
    def __init__(self, history_len=1000):
        self._raw_matrix = np.zeros(len(QUILT_SCHEMA), dtype=np.float32)
        self._raw_matrix[QUILT_SCHEMA["speed"].index] = 1.5 # Baseline troll speed
        
        # Thread safety locks for memory-mapped shared array
        self._lock = threading.Lock()
        
        # Memory-Mapped Write-Ahead Log (WAL) simulation using a fixed thread-safe ring buffer
        self.history_len = history_len
        self.wal_buffer = deque(maxlen=history_len)
        
        # Track last update timestamps to detect sensor dropouts or stale data
        self.last_update_times = {key: time.time() for key in QUILT_SCHEMA}

    def update_hardware_cell(self, semantic_key: str, raw_value: float):
        """Asynchronously called by hardware parsers. Enforces safety clipping."""
        if semantic_key in QUILT_SCHEMA:
            mapping = QUILT_SCHEMA[semantic_key]
            clipped = np.clip(raw_value, mapping.min_physical, mapping.max_physical)
            
            with self._lock:
                self._raw_matrix[mapping.index] = clipped
                self.last_update_times[semantic_key] = time.time()

    def commit_state_to_wal(self):
        """Snapshots current state vector into historical log queue at a fixed 10Hz clock tick."""
        with self._lock:
            snapshot = np.copy(self._raw_matrix)
            self.wal_buffer.append((time.time(), snapshot))

    def get_agent_tensor_feed(self) -> torch.Tensor:
        """Min-max normalizes current shared state into an agent-ready [-1.0, 1.0] tensor."""
        with self._lock:
            current_state = np.copy(self._raw_matrix)
            
        normalized = np.zeros_like(current_state)
        for key, mapping in QUILT_SCHEMA.items():
            val = current_state[mapping.index]
            span = mapping.max_physical - mapping.min_physical
            if span == 0:
                normalized[mapping.index] = 0.0
                continue
            normalized[mapping.index] = 2.0 * ((val - mapping.min_physical) / span) - 1.0
            
        return torch.tensor(normalized, dtype=torch.float32)

    def get_human_telemetry_string(self) -> str:
        with self._lock:
            current_state = np.copy(self._raw_matrix)
        lines = ["=== QUILT SYSTEM BROADCAST ==="]
        for key, mapping in QUILT_SCHEMA.items():
            val = current_state[mapping.index]
            lines.append(f"  [{mapping.index}] {mapping.name:<25}: {val:>7.2f} {mapping.unit}")
        return "\n".join(lines)

# =====================================================================# MISSING COMPONENT 1 & 3: ASYNC INGESTION DAEMON & SIGNAL INTERPOLATOR# =====================================================================class NmeaHardwareIngestionDaemon:
    """
    Simulates listening to live serial lines (/dev/ttyUSB) and parsing NMEA 0183 strings.
    Handles data streaming at mismatched frequencies natively without stalling threads.
    """
    def __init__(self, state_matrix: QuiltVesselStateMatrix):
        self.state = state_matrix
        self.running = False

    def parse_nmea_sentence(self, sentence: str):
        """Decodes raw marine text logs and maps them immediately to internal matrix coordinates."""
        try:
            if sentence.startswith("$GPRMC"):
                # Example GPS string: $GPRMC,speed=1.5,status=A
                parts = sentence.split(",")
                speed = float(parts[1])
                self.state.update_hardware_cell("speed", speed)
                
            elif sentence.startswith("$APHDG"):
                # Example Compass heading error string: $APHDG,err=3.4
                parts = sentence.split(",")
                heading_err = float(parts[1])
                self.state.update_hardware_cell("heading_error", heading_err)
                
            elif sentence.startswith("$GPXTE"):
                # Example Cross-track error string: $GPXTE,err=0.45
                parts = sentence.split(",")
                xte = float(parts[1])
                self.state.update_hardware_cell("xte", xte)
        except (ValueError, IndexError):
            pass # Gracefully handle malformed strings or serial noise common in heavy seas

    def _hardware_bus_loop(self):
        """Simulates varying physical loop times (Compass @ 10Hz, GPS @ 1Hz)."""
        loop_counter = 0
        while self.running:
            time.sleep(0.1) # Base hardware clock cycle
            loop_counter += 1
            
            # Compass updates rapidly at 10Hz
            sim_heading_err = np.random.normal(0, 2.0) + (np.sin(time.time() * 0.5) * 4.0)
            self.parse_nmea_sentence(f"$APHDG,{sim_heading_err}")
            
            # Cross-track and GPS positions drift slowly, reporting at 1Hz
            if loop_counter % 10 == 0:
                sim_xte = np.random.normal(0, 0.5) + (np.cos(time.time() * 0.1) * 1.5)
                self.parse_nmea_sentence(f"$GPXTE,{sim_xte}")
                self.parse_nmea_sentence(f"$GPRMC,1.5")
                
            # Commit the synchronized slice directly to the WAL log
            self.state.commit_state_to_wal()

    def start(self):
        self.running = True
        self.thread = threading.Thread(target=self._hardware_bus_loop, daemon=True)
        self.thread.start()

# =====================================================================# MISSING COMPONENT 4: PHYSICAL TRUST-GATED ACTUATION WRAPPER# =====================================================================class ActuationSafetyCore:
    """
    Sits between the model output tensor and the actual hydraulic pump relays.
    Ensures safe operations, manual fallback thresholds, and emergency overrides.
    """
    def __init__(self, state_matrix: QuiltVesselStateMatrix):
        self.state = state_matrix
        self.trust_factor = 0.0 # Strict manual safety baseline on startup

    def process_and_route_command(self, agent_output_tensor: torch.Tensor, dumb_autopilot_val: float) -> float:
        """Blends calculations or completely drops out if safety thresholds break."""
        # 1. Check for stale data (Emergency Timeout System)
        now = time.time()
        for key, last_seen in self.state.last_update_times.items():
            if key != "target_pump" and (now - last_seen) > 3.0:
                # If GPS or Compass stalls for more than 3 seconds, instantly drop to 0
                self.trust_factor = 0.0
                print(f"CRITICAL WARNING: Sensor '{key}' timeout. Reverting to manual control.")
        
        # 2. Denormalize the model's tensor back into raw hydraulic voltage values
        # Index 5 maps to the 'target_pump' boundary mapping [-1.0, 1.0]
        agent_scalar = agent_output_tensor[5].item()
        
        # 3. Apply the blended trust arbitration equation
        final_pump_command = ((1.0 - self.trust_factor) * dumb_autopilot_val) + (self.trust_factor * agent_scalar)
        
        # 4. Write back to the state matrix so the system monitors its own output
        self.state.update_hardware_cell("target_pump", final_pump_command)
        return final_pump_command
