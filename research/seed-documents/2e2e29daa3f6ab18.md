## Implementation Blueprint for Part 3: The Automated Git-Ratchet
This pipeline script executes your background training tasks, monitors model execution, and handles git-driven rollback mechanics automatically.

# auto_ratchet_runner.pyimport subprocessimport jsonimport osimport re
REGISTRY_FILE = "model_registry.json"TARGET_SCRIPT = "train.py"
def load_vessel_baseline_score() -> float:
    """Reads the current best tracking error baseline achieved by previous models."""
    if os.path.exists(REGISTRY_FILE):
        try:
            with open(REGISTRY_FILE, "r") as f:
                data = json.load(f)
                return float(data.get("best_score", float('inf')))
        except (json.JSONDecodeError, ValueError):
            pass
    # Baseline fallback score representing unoptimized dumb autopilot performance
    return 15.4000
def update_vessel_baseline_score(new_score: float):
    """Saves the newly established performance record to the registry."""
    with open(REGISTRY_FILE, "w") as f:
        json.dump({"best_score": new_score, "last_updated": os.getenv("USER", "boat_agent")}, f, indent=2)
def run_git_command(args: list[str]) -> str:
    """Helper process execution function to interact cleanly with local repository structures."""
    result = subprocess.run(["git"] + args, capture_output=True, text=True, check=True)
    return result.stdout.strip()
def execute_autoresearch_evaluation() -> float:
    """
    Launches the training loop process. Parses the terminal outputs 
    to extract the objective model track performance rating.
    """
    print(f"[Pipeline Execution] Launching background optimization loop pass over {TARGET_SCRIPT}...")
    try:
        # Run the training script with a hard 5-minute safety execution timeout ceiling
        result = subprocess.run(["python3", TARGET_SCRIPT], capture_output=True, text=True, timeout=300)
        
        if result.returncode != 0:
            print(f"  [Execution Failure] Script crashed with error status: {result.returncode}")
            print(result.stderr)
            return float('inf') # Treat compilation errors or logic crashes as maximum performance failures
            
        # Parse stdout looking for our designated scoring key
        match = re.search(r"OBJECTIVE_EFFICIENCY_SCORE:\s*([\d\.]+)", result.stdout)
        if match:
            return float(match.group(1))
        else:
            print("  [Parsing Error] Script executed but did not output an OBJECTIVE_EFFICIENCY_SCORE value line.")
            return float('inf')
            
    except subprocess.TimeoutExpired:
        print("  [Execution Failure] Script exceeded safe 5-minute training window threshold. Aborting run.")
        return float('inf')
def run_ratchet_cycle():
    """Evaluates workspace modifications, executing commits or rolling back bad code blocks."""
    best_historical_score = load_vessel_baseline_score()
    print(f"\n[Current Tracking State] Active Baseline Target Score: {best_historical_score:.4f}")
    
    # 1. Run the script optimization pass to evaluate the agent's code change
    candidate_score = execute_autoresearch_evaluation()
    print(f"[Evaluation Results] Candidate Model Architectural Score: {candidate_score:.4f}")
    
    # 2. Check if the newly proposed network architecture keeps a straighter course line
    if candidate_score < best_historical_score:
        print(">>> SUCCESS: Performance leap confirmed. Engaging Git Ratchet Mechanism.")
        
        # Update our registry configuration metric file
        update_vessel_baseline_score(candidate_score)
        
        try:
            # Stage changed logic files and record the update dynamically
            run_git_command(["add", TARGET_SCRIPT, REGISTRY_FILE])
            commit_msg = f"AutoResearch: Optimization milestone achieved. Track Score reduced to {candidate_score:.4f}"
            run_git_command(["commit", "-m", commit_msg])
            print(f"[Git Status] Staged modifications committed successfully: '{commit_msg}'")
        except subprocess.CalledProcessError as e:
            print(f"[Git Error] Failed to complete commit phase: {e.stderr}")
            
    else:
        print(">>> REGRESSION DETECTED: Code change degraded steering stability or crashed. Rolling back modifications...")
        try:
            # Execute a hard checkout of the code file to instantly scrub bad model code
            run_git_command(["checkout", "HEAD", "--", TARGET_SCRIPT])
            print(f"[Git Status] Workspace successfully cleaned. Reverted {TARGET_SCRIPT} back to previous stable baseline.")
        except subprocess.CalledProcessError as e:
            print(f"[Git Error] Critical failure during checkout correction loop: {e.stderr}")
if __name__ == "__main__":
    # Ensure system folder context is an initialized git repository before processing tasks
    try:
        run_git_command(["status"])
    except subprocess.CalledProcessError:
        print("[Repository Notice] Workspace folder is not currently a git repo. Initializing locally...")
        run_git_command(["init"])
        run_git_command(["add", "."])
        run_git_command(["commit", "-m", "Initial vessel core environment baseline commit"])
        
    run_ratchet_cycle()
