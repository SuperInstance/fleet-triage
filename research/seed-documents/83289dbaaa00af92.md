  [ STATE ]    ───►    [ FILTER ]   ───►    [ FERRE ]

## Building the Cellular Substrate
This architecture can be built cleanly using a low-overhead array structure. The script below constructs a functional Quilt Substrate. It models cellular pressure fields, lets data filter smoothly into neighboring spaces, and uses vector aggregation to let the system converge on an answer using pure geometric balance.

# quilt_logic_core.pyimport osimport jsonimport math
# Initialize generation folders
os.makedirs("generated", exist_ok=True)
class QuiltCell:
    def __init__(self, x: int, y: int):
        self.x = x
        self.y = y
        self.state_potential = 0.0  # Stored structural charge
        self.filter_resistance = 0.4  # Natural threshold gate barrier
        self.vector_direction = [0.0, 0.0]  # Structural pressure orientation

    def calculate_round_tension(self, target_x: float, target_y: float):
        """Calculates smooth geometric distance, avoiding sharp step functions."""
        dx = target_x - self.x
        dy = target_y - self.y
        distance = math.sqrt(dx**2 + dy**2)
        
        # Continuous vector assignment to maintain perfect geometric roundness
        if distance > 0:
            self.vector_direction = [dx / distance, dy / distance]
        return distance
class QuiltSubstrate:
    def __init__(self, size: int = 5):
        self.size = size
        self.grid = {(x, y): QuiltCell(x, y) for x in range(size) for y in range(size)}

    def inject_force(self, x: int, y: int, potential: float):
        """Injects raw energy directly into a specific point on the fabric."""
        if (x, y) in self.grid:
            self.grid[(x, y)].state_potential = potential

    def execute_filter_and_ferre_cycle(self):
        """Executes a single natural flow pass across the fabric."""
        cloned_potentials = {coords: cell.state_potential for coords, cell in self.grid.items()}
        
        # The Filter Phase: Data leaks into neighboring cells based on friction boundaries
        for (x, y), cell in self.grid.items():
            neighbors = [(x+1, y), (x-1, y), (x, y+1), (x, y-1)]
            for nx, ny in neighbors:
                if (nx, ny) in self.grid:
                    neighbor_cell = self.grid[(nx, ny)]
                    pressure_differential = cloned_potentials[(x, y)] - cloned_potentials[(nx, ny)]
                    
                    # Data only moves if the pressure breaks past the filter's resistance threshold
                    if pressure_differential > cell.filter_resistance:
                        leakage = (pressure_differential - cell.filter_resistance) * 0.25
                        cell.state_potential -= leakage
                        neighbor_cell.state_potential += leakage

    def extract_equilibrium_state(self) -> dict:
        """The Ferre Phase: Extracts the final coordinate where geometric balance is achieved."""
        total_potential = 0.0
        weighted_x = 0.0
        weighted_y = 0.0
        
        telemetry = []
        for (x, y), cell in self.grid.items():
            if cell.state_potential > 0.05:
                total_potential += cell.state_potential
                weighted_x += x * cell.state_potential
                weighted_y += y * cell.state_potential
                
                telemetry.append({
                    "coords": [x, y],
                    "potential": round(cell.state_potential, 3)
                })

        # Calculate the exact center of balance across the continuous field
        center_of_mass = [weighted_x / total_potential, weighted_y / total_potential] if total_potential > 0 else [0, 0]
        
        return {
            "center_of_equilibrium": [round(center_of_mass[0], 2), round(center_of_mass[1], 2)],
            "total_field_potential": round(total_potential, 3),
            "active_cells": telemetry
        }
