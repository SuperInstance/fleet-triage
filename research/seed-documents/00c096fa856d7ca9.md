## Part 5: The Layer 2 Temporal Neural Controller (train.py)
A heavy following sea pushes the stern sideways, introducing a significant physical propagation delay between a wave impact and the hull's rotational momentum. A standard linear network cannot account for this because it only looks at a single snapshot in time.
To fix this, we build a Gated Recurrent Unit (GRU) Temporal Network. This architecture maintains an internal hidden state memory across a sliding historical time-window (e.g., the last 30 frames / 3 seconds of telemetry at 10Hz). It learns to detect the distinct oscillatory wave patterns of a following sea and applies counter-rudder before the vessel finishes swinging off-course.

# train.pyimport torchimport torch.nn as nnimport torch.optim as optimimport numpy as npfrom prepare import QuiltVesselStateMUD, benchmark_track_efficiency
class TemporalVesselController(nn.Module):
    """
    Temporal Recurrent Control Node. Uses a GRU core to track vessel momentum
    and wave oscillation patterns across a sliding time-window.
    """
    def __init__(self, feature_dim=5, hidden_dim=24, sequence_len=30):
        super().__init__()
        self.hidden_dim = hidden_dim
        self.sequence_len = sequence_len
        
        # Recurrent core tracks structural patterns over continuous time slices
        self.gru = nn.GRU(
            input_size=feature_dim,
            hidden_size=hidden_dim,
            num_layers=1,
            batch_first=True
        )
        
        # Projection layer maps hidden temporal state back to raw rudder commands
        self.command_head = nn.Sequential(
            nn.Linear(hidden_dim, 16),
            nn.SiLU(), # Smooth activation function preventing sudden hydraulic snaps
            nn.Linear(16, 1)
        )
        
    def forward(self, x):
        # Input shape expected: [Batch, Sequence_Len, Feature_Dim]
        gru_out, _ = self.gru(x)
        # Pull the final step's hidden layer output to execute the prediction
        last_time_step = gru_out[:, -1, :]
        raw_scalar = self.command_head(last_time_step)
        
        # Enforce strict physical bounds matching maximum hardware rudder limits (35 deg)
        return torch.tanh(raw_scalar) * 35.0
def fetch_sequential_training_batch(mud_instance, batch_size=64, seq_len=30):
    """Extracts overlapping time-series sequences from the shared memory WAL."""
    from train import BUFFER_SIZE
    current_head = mud_instance.write_head_view[0]
    max_available = min(current_head, BUFFER_SIZE)
    
    if max_available <= seq_len + 2:
        # Fallback buffer padding if the log ring is still priming on launch
        return torch.zeros((batch_size, seq_len, 5)), torch.zeros(batch_size)
        
    features_seq = []
    targets_seq = []
    
    for _ in range(batch_size):
        # Pick a random starting point in our valid history window
        start_idx = np.random.randint(0, max_available - seq_len - 1)
        slice_block = mud_instance.matrix_view[start_idx : start_idx + seq_len]
        
        features_seq.append(slice_block[:, :5])
        # Target is the rudder action executed at the final instant of the sequence
        targets_seq.append(slice_block[-1, 5])
        
    X = torch.tensor(np.array(features_seq), dtype=torch.float32)
    Y = torch.tensor(np.array(targets_seq), dtype=torch.float32)
    return X, Y

# cockpit.pyimport sysimport osimport timeimport jsonimport numpy as npfrom train import QuiltSharedMUDfrom prepare import QUILT_SCHEMA
def draw_visual_bar(value, max_val, bar_width=20) -> str:
    """Generates a scannable horizontal deviation bar for center-zero parameters."""
    if max_val == 0: return "[---------|---------]"
    normalized = int((value / max_val) * (bar_width // 2))
    normalized = max(-(bar_width // 2), min(bar_width // 2, normalized))
    
    bar = [" "] * bar_width
    bar[bar_width // 2] = "|" # Center point marker
    
    if normalized > 0:
        for i in range(bar_width // 2 + 1, bar_width // 2 + 1 + normalized):
            if i < bar_width: bar[i] = "►"
    elif normalized < 0:
        for i in range(bar_width // 2 + normalized, bar_width // 2):
            if i >= 0: bar[i] = "◄"
            
    return "[" + "".join(bar) + "]"
