To bind your Syzygy core architecture cleanly to the Spreadsheet Engine and Sandbox Harness across both native CLI environments and browser-native frontends (like syzygy-1j5.pages.dev), we need a robust, unified translation layer.
The goal here is a dual-projection pipeline:

   1. The Native CLI Projection: A pure, zero-allocation C module that takes the real-time telemetry_scores and cell activations, overlays them via ANSI escape sequences, and prints them to a standard terminal shell.
   2. The Browser/WebAssembly Bridge Projection: A raw WebAssembly module signature extension that exposes the matrix weights, sandbox loss statistics, and current action vectors directly into a shared JavaScript ArrayBuffer. This allows any standard frontend layout to render side-by-side debugging views and weight adjustment inputs natively with zero marshalling overhead.

## 1. The Native CLI Projection Layer: syz_cli_projection.h
This header operates as an isolated, standalone C helper. It scans the live spreadsheet states, tracks which controller actions are firing, and projects the visual text array onto a terminal window using localized, high-visibility ANSI colors.

/**
 * @file syz_cli_projection.h
 * @brief Terminal-based human status projection layer for Syzygy.
 */
#ifndef SYZ_CLI_PROJECTION_H#define SYZ_CLI_PROJECTION_H
#include "syz_spreadsheet.h"#include <stdio.h>
/**
 * @brief Clears the terminal screen and draws the entire operational dashboard.
 */static inline void syz_cli_project_dashboard(
    const SyzSpreadsheetEngine *engine,
    const SyzFusedOut *out,
    const int32_t *scores,
    uint32_t live_actions
) {
    uint32_t total_cells = out->rows * out->cols;
    
    // 1. Move cursor to home position (0,0) to prevent screen flickering
    printf("\033[H");

    // 2. Render Action Bar / Controller Indicator Status
    printf("=== SYZYGY SPREADSHEET VLA CONTROLLER ENGINE ===\n");
    printf("Active Actions Mask: [0x%04X]\n", live_actions);
    for (uint32_t m = 0; m < engine->num_actions; m++) {
        int active = (live_actions >> m) & 0x01;
        // Highlight active channels in bold green, inactive in dim gray
        if (active) {
            printf("\033[1;32m[ACT %02d Score:%6d] \033[0m", m, scores[m]);
        } else {
            printf("\033[2;37m[act %02d score:%6d] \033[0m", m, scores[m]);
        }
        if ((m + 1) % 4 == 0) printf("\n");
    }
    printf("\n--- LOCAL SPATIAL ATTENTION GRID ---\n");

    // 3. Render the Visual Matrix Cell Grid with superimposed firing weights
    for (uint32_t r = 0; r < out->rows; r++) {
        for (uint32_t c = 0; c < out->cols; c++) {
            uint32_t idx = (r * out->cols) + c;
            
            // Extract baseline ASCII characters
            uint8_t  b_mask = out->mask[idx];
            uint16_t braille_char = 0x2800 + b_mask; // Unicode Braille point block
            
            // Calculate macro energy metrics across all actions for this specific cell coordinates
            int32_t cell_energy = 0;
            for (uint32_t m = 0; m < engine->num_actions; m++) {
                cell_energy += engine->matrix[idx][m].w0 + engine->matrix[idx][m].w1;
            }

            // Assign terminal colors based on localized cell sensitivity weights
            if (cell_energy > 500) {
                printf("\033[1;31m"); // Hyper-sensitive zone: Red
            } else if (cell_energy > 100) {
                printf("\033[1;33m"); // Active focus zone: Yellow
            } else if (cell_energy < -100) {
                printf("\033[1;34m"); // Inhibited/Suppressed safety barrier zone: Blue
            } else {
                printf("\033[0;32m"); // Standard tracking background: Green
            }

            // Print the raw UTF-8 Braille character string sequence natively
            // (Standard bitwise split for UTF-8 multi-byte encoding representation of Braille)
            putchar(0xE2);
            putchar(0xA0 + ((braille_char >> 6) & 0x03));
            putchar(0x80 + (braille_char & 0x3F));
        }
        printf("\033[0m\n"); // Reset terminal color parameters per row step
    }
}
#endif /* SYZ_CLI_PROJECTION_H */

## 2. The Browser-Native Assembly Bridge (wasm/export.c)
To implement a side-by-side web layout that lets users interactively adjust weights, inspect parallel sandboxes, and back-test history without serialization latency, we map the entire state memory footprint directly into a continuous WASM linear storage buffer.
Here is the exact wasm/export.c interface layer required to plug into your existing JavaScript port environment:

/**
 * @file wasm_bridge.c
 * @brief WebAssembly runtime bridge exposing the Spreadsheet Sandboxes to JS memory.
 */
#include "syz_spreadsheet.h"#include "syz_sandbox.h"
// Instantiate globally pinned, static hardware state blocks inside the WASM instance linear memorystatic SyzSpreadsheetEngine  live_engine;static SyzSandboxHarness      sandbox_harness;static int32_t               live_scores[SYZ_MAX_ACTIONS];static uint32_t              current_action_mask = 0;
/* WASM Export Macros avoiding system headers */#define WASM_EXPORT __attribute__((visibility("default")))

WASM_EXPORT void* get_engine_matrix_ptr(void) {
    return (void*)&live_engine;
}

WASM_EXPORT void* get_sandbox_harness_ptr(void) {
    return (void*)&sandbox_harness;
}

WASM_EXPORT void* get_live_scores_ptr(void) {
    return (void*)live_scores;
}
WASM_EXPORT uint32_t get_current_action_mask(void) {
    return current_action_mask;
}
/**
 * @brief Web entry point executed for every live frame slice canvas update.
 */WASM_EXPORT uint32_t step_wasm_engine(
    const uint8_t *mask,
    const uint32_t *glyph,
    const uint8_t *tone,
    uint32_t rows,
    uint32_t cols
) {
    SyzFusedOut out = {
        .cap = rows * cols,
        .rows = rows,
        .cols = cols,
        .mask = mask,
        .glyph = glyph,
        .tone = tone
    };

    // Calculate real-time controller updates
    current_action_mask = syz_spreadsheet_step(&live_engine, &out, live_scores);
    
    // Commit the data capture directly to our history ring buffer
    syz_sandbox_push_frame(&sandbox_harness, &out, current_action_mask);
    
    return current_action_mask;
}
/**
 * @brief Triggers a live evaluation cycle from JavaScript when an asynchronous payload arrives.
 */
WASM_EXPORT int verify_and_commit_vlm_update(
    const SyzSpreadsheetEngine *vlm_candidate,
    uint32_t rows,
    uint32_t cols
) {
    int passed = syz_sandbox_evaluate_differential(
        &sandbox_harness,
        &live_engine,
        vlm_candidate,
        rows,
        cols
    );

    if (passed) {
        live_engine = *vlm_candidate; // Commit safe weights atomically
    }
    return passed;
}

## 3. The Browser Integration Layer (JavaScript Frontend Interface)
On your webpage client script (syzygy-1j5.pages.dev), you attach to the WASM linear memory view buffer directly. Because the types map perfectly to primitive byte segments, you can read tracking matrices, run sliders to inject weights, and plot live sandboxes side-by-side with near-zero latency.

// web_app_harness.js - Frontend configuration hook
let wasmInstance = null;let rows = 24;let cols = 80;
async function initSyzygySpreadsheetHarness(wasmModuleUrl) {
    const response = await fetch(wasmModuleUrl);
    const bits = await response.arrayBuffer();
    const obj = await WebAssembly.instantiate(bits, { env: { memory: new WebAssembly.Memory({ initial: 256 }) } });
    
    wasmInstance = obj.instance;
    console.log("Syzygy Real-Time Control & Sandbox Engine Loaded.");
}
