@dataclassclass SpreadsheetCell:
    cell_id: str  # e.g., "A1", "B4"
    row: int
    col: int
    potential: float = 0.0
    resistance: float = 0.4
    entropy: float = 0.0
    is_split: bool = False
    children_ids: List[str] = field(default_factory=list)
class IntegratedQuiltOS:
    def __init__(self, size: int = 4, split_threshold: float = 0.7):
        self.size = size
        self.split_threshold = split_threshold
        self.grid: Dict[Tuple[int, int], SpreadsheetCell] = {}
        self._initialize_spreadsheet_grid()

    def _initialize_spreadsheet_grid(self):
        """Builds a standard coordinate matrix mapping directly to spreadsheet addresses."""
        for r in range(self.size):
            for c in range(self.size):
                cell_name = f"{chr(65 + c)}{r + 1}"  # Maps 0,0 to A1, 1,1 to B2
                self.grid[(r, c)] = SpreadsheetCell(cell_id=cell_name, row=r, col=c)

    def inject_spreadsheet_force(self, cell_id: str, potential: float):
        """Locates a cell by its spreadsheet coordinate name and injects raw potential."""
        for cell in self.grid.values():
            if cell.cell_id == cell_id:
                cell.potential = potential
                print(f"[*] Force Injection -> Cell {cell_id} charged with potential: {potential}")

    def execute_substrate_flow_cycle(self):
        """Allows potentials to leak smoothly through row and column filters."""
        cloned_potentials = {coords: cell.potential for coords, cell in self.grid.items()}
        
        for (r, c), cell in self.grid.items():
            # Check immediate horizontal and vertical neighbors
            neighbors = [(r+1, c), (r-1, c), (r, c+1), (r, c-1)]
            for nr, nc in neighbors:
                if (nr, nc) in self.grid:
                    neighbor_cell = self.grid[(nr, nc)]
                    differential = cloned_potentials[(r, c)] - cloned_potentials[(nr, nc)]
                    
                    # The Filter Phase: Move data only if it breaks past resistance thresholds
                    if differential > cell.resistance:
                        leakage = (differential - cell.resistance) * 0.2
                        cell.potential -= leakage
                        neighbor_cell.potential += leakage

    def execute_jit_splitting_check(self):
        """Scans the active fields and fractally splits high-entropy spreadsheet zones."""
        for (r, c), cell in self.grid.items():
            # Calculate a local entropy scalar based on its potential field intensity
            if cell.potential > 0:
                cell.entropy = min(1.0, (cell.potential / 5.0) * 0.9)
            else:
                cell.entropy = 0.0

            # JIT Trigger: Split cell if it breaches the operational entropy gate
            if cell.entropy > self.split_threshold and not cell.is_split:
                print(f"[!] JIT SPLIT -> Entropy spike at {cell.cell_id} ({cell.entropy:.2f}). Generating nested child matrix.")
                cell.is_split = True
                
                # Generate 4 virtual sub-cells packed inside this spreadsheet cell's address space
                child_sub_names = [f"{cell.cell_id}_NW", f"{cell.cell_id}_NE", f"{cell.cell_id}_SW", f"{cell.cell_id}_SE"]
                cell.children_ids = child_sub_names
                cell.potential *= 0.5  # Energy disperses down into the new sub-tissue level

    def export_to_csv_spreadsheet(self, filepath: str = "generated/quilt_matrix.csv"):
        """Compiles the entire active geometric substrate out to a raw, flat-file CSV sheet."""
        with open(filepath, mode="w", newline="") as f:
            writer = csv.writer(f)
            # Write system data headers
            writer.writerow(["Spreadsheet Coordinate", "Row Index", "Column Index", "Potential Value", "Resistance Threshold", "Local Entropy", "JIT Split Status", "Sub-Cell Lattice Addresses"])
            
            for cell in sorted(self.grid.values(), key=lambda x: (x.row, x.col)):
                writer.writerow([
                    cell.cell_id,
                    cell.row,
                    cell.col,
                    round(cell.potential, 3),
                    cell.resistance,
                    round(cell.entropy, 3),
                    "SPLIT_ACTIVE" if cell.is_split else "STATIC",
                    " | ".join(cell.children_ids) if cell.children_ids else "NONE"
                ])
        print(f"\n[+] Spreadsheet successfully compiled and exported to: {filepath}")
