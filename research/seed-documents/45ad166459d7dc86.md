class QuiltVesselStateMUD:
    """
    General-purpose memory-mapped simulation of the vessel's live state tree.
    Acts as the single source of truth for both live operations and training loops.
    """
    def __init__(self, buffer_size=10000):
        self.buffer_size = buffer_size
        # State vector: [Heading Error, Cross-Track Error, Pitch, Roll, Speed, Actual Rudder]
        self.state_memory = np.zeros((buffer_size, 6), dtype=np.float32)
        self.pointer = 0
        self._populate_historical_season_logs()
        
    def _populate_historical_season_logs(self):
        """Simulates historical telemetry populated continuously by the bridge daemon."""
        np.random.seed(42)
        # Generate realistic trolling variables at 1.5 knots in heavy swell
        heading_err = np.random.normal(0, 3.5, self.buffer_size)
        xte = np.random.normal(0, 1.2, self.buffer_size)
        pitch = np.sin(np.linspace(0, 50, self.buffer_size)) * 2.0
        roll = np.cos(np.linspace(0, 50, self.buffer_size)) * 4.0
        speed = np.ones(self.buffer_size) * 1.5
        
        # What the human/dumb autopilot did to correct these inputs
        actual_rudder = (heading_err * 1.6) + (xte * 0.75) + (roll * 0.1)
        actual_rudder = np.clip(actual_rudder, -35.0, 35.0)
        
        self.state_memory[:, 0] = heading_err
        self.state_memory[:, 1] = xte
        self.state_memory[:, 2] = pitch
        self.state_memory[:, 3] = roll
        self.state_memory[:, 4] = speed
        self.state_memory[:, 5] = actual_rudder

    def stream_synced_tensors(self, batch_size=64):
        """Yields perfectly aligned feature/target matrices for background learning."""
        indices = np.random.randint(0, self.buffer_size - 1, batch_size)
        batch = self.state_memory[indices]
        
        features = torch.tensor(batch[:, :5], dtype=torch.float32)
        targets = torch.tensor(batch[:, 5], dtype=torch.float32)
        return features, targets
def benchmark_track_efficiency(model_node, state_mud):
    """Calculates objective tracking accuracy and hydraulic efficiency metrics."""
    model_node.eval()
    # Test against a frozen section of the state tree
    eval_data = torch.tensor(state_mud.state_memory[-1000:, :5], dtype=torch.float32)
    true_rudder = torch.tensor(state_mud.state_memory[-1000:, 5], dtype=torch.float32)
    
    with torch.no_grad():
        predicted_rudder = model_node(eval_data).squeeze()
        # Objective Metric: Minimize deviation while penalizing high-frequency rudder chatter
        tracking_loss = torch.mean((predicted_rudder - true_rudder) ** 2).item()
        hydraulic_wear = torch.mean(torch.abs(predicted_rudder[1:] - predicted_rudder[:-1])).item()
        
    return tracking_loss + (0.8 * hydraulic_wear)

class HardwareTrustGate:
    def __init__(self, dumb_autopilot_sys, neural_node_sys):
        self.dumb_sys = dumb_autopilot_sys
        self.neural_sys = neural_node_sys
        self.trust_factor = 0.0  # 0.0 = 100% Dumb Autopilot, 1.0 = 100% Neural Node
        
    def calculate_blended_rudder(self, current_state_vector):
        """Smoothly blends commands based on the captain's confidence level."""
        dumb_command = self.dumb_sys.get_rudder_angle(current_state_vector)
        neural_command = self.neural_sys.predict_rudder(current_state_vector)
        
        # Linear interpolation between systems ensures no sudden physical shifts
        final_rudder_angle = ((1.0 - self.trust_factor) * dumb_command) + (self.trust_factor * neural_command)
        return np.clip(final_rudder_angle, -35.0, 35.0)
