## Part 1: Python Morton (Z-Order) Spatial Encoder
This encoder takes your raw 3D coordinates (X, Y, Z) within the 512³ grid and interleaves their bits to generate a single 27-bit Morton code. This ensures that voxels that are physically close in 3D space are stored next to each other in memory, maximizing performance when the compute shader loads them.

import torchimport torch.nn as nnimport numpy as np
def encode_morton_3d(x: int, y: int, z: int) -> int:
    """
    Interleaves the 9 bits of X, Y, and Z into a single 27-bit Morton code
    for optimized spatial cache locality on the GPU.
    """
    def spread_bits(n: int) -> int:
        n = (n | (n << 16)) & 0x0000FF0000FF
        n = (n | (n << 8))  & 0x00F00F00F00F
        n = (n | (n << 4))  & 0x30C30C30C30C
        n = (n | (n << 2))  & 0x924924924924
        return n & 0x1FF249249249  # Protect 9 bits spread across 27 spaces
    
    return (spread_bits(x) << 2) | (spread_bits(y) << 1) | spread_bits(z)
def pack_voxel_token(x: int, y: int, z: int, feature_id: int) -> int:
    """
    Packs a 512^3 coordinate and a 5-bit feature ID into a single 32-bit token.
    Lower 5 bits: Feature ID
    Upper 27 bits: Morton coded spatial index
    """
    morton_code = encode_morton_3d(x, y, z)
    # Ensure bounds: morton fits in 27 bits (max 134,217,727), feature_id fits in 5 bits (max 31)
    packed_token = ((morton_code & 0x7FFFFFF) << 5) | (feature_id & 0x1F)
    return packed_token

## Part 2: Downstream Cross-Attention Mapping Layer
Once the WGSL shader finishes the Fused Projection Pass and outputs the dense, coordinate-injected matrix slice, this PyTorch layer acts as the bridge. It performs a fast linear map and maps the sparse voxel sequence lengths directly into your transformer's token space using Multi-Head Cross-Attention.

class VoxelCrossAttentionBridge(nn.Module):
    def __init__(self, voxel_dim=64, transformer_dim=512, n_heads=8):
        super().__init__()
        self.n_heads = n_heads
        
        # Aligns the WGSL shader output dimension (64) to your target Transformer dimension (512)
        self.proj_voxel = nn.Linear(voxel_dim, transformer_dim)
        
        # Core Cross-Attention mechanism linking voxel state spaces to latent target vectors
        self.cross_attn = nn.MultiheadAttention(
            embed_dim=transformer_dim, 
            num_heads=n_heads, 
            batch_first=True
        )
        
        self.layer_norm = nn.LayerNorm(transformer_dim)
        self.ffn = nn.Sequential(
            nn.Linear(transformer_dim, transformer_dim * 4),
            nn.ReLU(),
            nn.Linear(transformer_dim * 4, transformer_dim)
        )

    def forward(self, shader_latents, query_tokens):
        """
        Args:
            shader_latents (Tensor): Matrix out of the WGSL shader [Batch, Total_Active_Voxels, Voxel_Dim]
            query_tokens (Tensor): Active model target/query sequence [Batch, Target_Seq_Len, Transformer_Dim]
        Returns:
            Tensor: Aligned spatial structural embeddings [Batch, Target_Seq_Len, Transformer_Dim]
        """
        # 1. Align dimensions from shader layout to model architecture
        projected_voxels = self.proj_voxel(shader_latents)
        
        # 2. Compute spatial context vectors via Multi-Head Cross Attention
        # Queries come from the downstream model; Keys/Values come from the spatial shader stream
        attn_out, _ = self.cross_attn(
            query=query_tokens, 
            key=projected_voxels, 
            value=projected_voxels
        )
        
        # 3. Apply standard optimization residual & feed-forward routing blocks
        x = self.layer_norm(query_tokens + attn_out)
        out = self.layer_norm(x + self.ffn(x))
        return out
