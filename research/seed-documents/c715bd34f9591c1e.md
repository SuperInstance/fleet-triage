# physical_bridge.pyimport sysimport osimport timeimport threadingimport torchimport numpy as npfrom main_vessel_core import IntegratedVesselEnginefrom prepare import QUILT_SCHEMA
class LowLevelHardwareSerialBridge:
    """
    Manages physical I/O streams and asynchronous hardware parsing interfaces.
    Bridges live serial lines (/dev/ttyUSB) directly to the memory-mapped State MUD.
    """
    def __init__(self, engine_instance: IntegratedVesselEngine, port_path="/dev/ttyUSB0"):
        self.engine = engine_instance
        self.port_path = port_path
        self.running = False
        self.reader_thread = None
        
        # Buffer to aggregate fragments from rough serial lines
        self.serial_accumulator = ""

    def process_raw_line(self, line: str):
        """
        Parses raw text sentences from instruments, extracting critical telemetry
        metrics and mapping them instantly to designated matrix coordinates.
        """
        clean_line = line.strip()
        if not clean_line.startswith("$"):
            return

        try:
            # 1. Parse GPS Cross-Track Error sentences
            if clean_line.startswith("$GPXTE") or "XTE" in clean_line:
                # Format: $GPXTE,A,A,0.12,L,N*offset
                parts = clean_line.split(",")
                raw_xte = float(parts[3])
                # Convert 'L' (Left of track) to negative error boundary representation
                if len(parts) > 4 and parts[4] == "L":
                    raw_xte = -raw_xte
                self.engine.local_state.update_hardware_cell("xte", raw_xte)

            # 2. Parse Autopilot/Compass Heading sentences
            elif clean_line.startswith("$APHDG") or "HDG" in clean_line:
                # Format: $APHDG,Heading,Dev,Dir,Var,Dir
                parts = clean_line.split(",")
                current_heading = float(parts[1])
                # Calculate heading error relative to the current dial set-point
                target_heading = 180.0  # Placeholder target course heading
                heading_err = current_heading - target_heading
                
                # Normalize error rotation boundaries between [-180, 180]
                heading_err = (heading_err + 180) % 360 - 180
                self.engine.local_state.update_hardware_cell("heading_error", heading_err)

            # 3. Parse Custom Analogue Rudder Follower Potentiometer metrics
            elif clean_line.startswith("$PBBRA") or "RUD" in clean_line:
                # Format: Custom proprietary text sentence string: $PBBRA,rudder_deg
                parts = clean_line.split(",")
                rudder_pos = float(parts[1])
                self.engine.local_state.update_hardware_cell("rudder_angle", rudder_pos)

        except (ValueError, IndexError) as e:
            # Log parsing errors safely without interrupting the primary loop execution context
            pass

    def _io_execution_worker(self):
        """
        Asynchronous background read loop. Handles raw file descriptor/device 
        streams without blocking or causing data drops.
        """
        print(f"[Hardware Interface] Opening serial communications link over target path: {self.port_path}")
        
        # Safe fallback routine if executing in an offline development sandbox environment
        if not os.path.exists(self.port_path):
            print(f"[Warning] Physical port path target '{self.port_path}' not discovered. Initializing software emulator deck...")
            self._execute_software_emulation_stream()
            return

        try:
            with open(self.port_path, "r", encoding="ascii", errors="ignore") as serial_device:
                while self.running:
                    # Non-blocking character block chunk evaluation reads
                    chunk = serial_device.read(64)
                    if not chunk:
                        time.sleep(0.01)
                        continue
                        
                    self.serial_accumulator += chunk
                    while "\n" in self.serial_accumulator:
                        line, self.serial_accumulator = self.serial_accumulator.split("\n", 1)
                        self.process_raw_line(line)
        except Exception as err:
            print(f"[Hardware Error] Serial device stream interrupted: {err}")
            self.running = False

    def _execute_software_emulation_stream(self):
        """Provides simulated data streams when developing offline away from the boat."""
        sim_counter = 0
        while self.running:
            time.sleep(0.1) # 10Hz clock cycle emulation pulse
            sim_counter += 1
            
            # Generate simulated wave impacts and steering drift metrics
            wave_oscillation = np.sin(time.time() * 0.4) * 3.5
            rudder_feedback = self.engine.local_state._raw_matrix[QUILT_SCHEMA["rudder_angle"].index]
            
            sim_hdg_err = wave_oscillation - (rudder_feedback * 0.5) + np.random.normal(0, 0.5)
            self.process_raw_line(f"$APHDG,{180.0 + sim_hdg_err},,,")
            
            if sim_counter % 10 == 0:
                sim_xte = np.sin(time.time() * 0.05) * 1.2
                self.process_raw_line(f"$GPXTE,,,{abs(sim_xte)},{'L' if sim_xte < 0 else 'R'},N")
                self.process_raw_line(f"$PBBRA,{rudder_feedback + np.random.normal(0, 0.2)}")

    def start_bridge_thread(self):
        self.running = True
        self.reader_thread = threading.Thread(target=self._io_execution_worker, daemon=True)
        self.reader_thread.start()
        print("[System Status] Low-level asynchronous hardware link layer online.")

# =====================================================================# THE PHYSICAL ACTUATION LAYER CONTROL TERMINAL# =====================================================================class HydraulicPumpRelayController:
    """
    Handles physical hardware output logic. Translates model voltage calculations 
    back into discrete, directional relay pulses.
    """
    def __init__(self, relay_device_path="/dev/hardware_relay_output"):
        self.relay_device_path = relay_device_path

    def output_voltage_to_relays(self, target_command_scalar: float):
        """
        Maps a continuous command scalar bounded between [-1.0, 1.0] 
        to active physical commands (e.g., Pump Left / Pump Right).
        """
        # Apply a small deadband buffer to protect steering components from constant switching
        DEADBAND = 0.05
        
        if abs(target_command_scalar) < DEADBAND:
            # Neutral state: Maintain current position, open all pump solenoids
            sys_action = "PUMP_HOLD_NEUTRAL"
        elif target_command_scalar > 0:
            # Positive value: Route voltage to the starboard pump relay
            sys_action = f"PUMP_RIGHT_SOLENOID_ON [Duty: {abs(target_command_scalar) * 100:.1f}%]"
        else:
            # Negative value: Route voltage to the port pump relay
            sys_action = f"PUMP_LEFT_SOLENOID_ON [Duty: {abs(target_command_scalar) * 100:.1f}%]"
            
        # Real-world deployment targets write directly to GPIO sysfs pins or serial DAC boards here
        # Example: os.write(self.gpio_fd, b"1")
        return sys_action
